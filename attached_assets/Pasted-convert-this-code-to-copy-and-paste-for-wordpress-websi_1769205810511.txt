import React, { useState } from 'react';
import { Calculator, Users, Package, DollarSign, Zap, Code, Cpu } from 'lucide-react';

const OdooQuoteCalculator = () => {
  const [users, setUsers] = useState(10);
  const [plan, setPlan] = useState('standard');
  const [implementation, setImplementation] = useState('none');
  
  // Term discounts for each billing option (plan and implementation separate)
  const [termDiscounts, setTermDiscounts] = useState({
    monthly: { plan: 0, impl: 0 },
    '1year': { plan: 0, impl: 0 },
    '2year': { plan: 0, impl: 0 },
    '3year': { plan: 0, impl: 0 },
    '4year': { plan: 0, impl: 0 },
    '5year': { plan: 0, impl: 0 }
  });

  // Term toggles for quote inclusion
  const [selectedTerms, setSelectedTerms] = useState({
    monthly: false,
    '1year': true,
    '2year': false,
    '3year': false,
    '4year': false,
    '5year': false
  });

  // Pricing data
  const pricing = {
    standard: {
      monthly: { year1: 31.10, year2plus: 38.90 },
      yearly: { year1: 24.90, year2plus: 31.10 }
    },
    custom: {
      monthly: { year1: 61.10, year2plus: 76.20 },
      yearly: { year1: 49.00, year2plus: 61.00 }
    }
  };

  const implementations = {
    express: { hours: 4, price: 540 },
    starter: { hours: 25, price: 3600 },
    basic: { hours: 50, price: 7000 },
    standard: { hours: 100, price: 12500 },
    custom: { hours: 200, price: 25000 }
  };

  const calculateTermQuote = (termKey, planDiscount, implDiscount) => {
    const planPricing = pricing[plan];
    
    if (termKey === 'monthly') {
      // Monthly billing
      const monthlyRate = planPricing.monthly.year1;
      const monthlySoftwareCost = monthlyRate * users;
      const planDiscountAmount = monthlySoftwareCost * (planDiscount / 100);
      const finalMonthlyCost = monthlySoftwareCost - planDiscountAmount;
      
      // Implementation
      const implCost = implementation !== 'none' ? implementations[implementation].price : 0;
      const implDiscountAmount = implCost * (implDiscount / 100);
      const finalImplCost = implCost - implDiscountAmount;
      
      return {
        termLabel: 'Monthly',
        softwareCost: finalMonthlyCost,
        implementationCost: finalImplCost,
        planDiscountAmount,
        implDiscountAmount,
        isMonthly: true
      };
    } else {
      // Yearly billing - calculate total cost of ownership
      const years = parseInt(termKey.replace('year', ''));
      let totalSoftwareCost = 0;
      
      // Year 1: first year rate * 12 months
      totalSoftwareCost += planPricing.yearly.year1 * users * 12;
      
      // Years 2+: year 2+ rate * 12 months * remaining years
      if (years > 1) {
        totalSoftwareCost += planPricing.yearly.year2plus * users * 12 * (years - 1);
      }
      
      const planDiscountAmount = totalSoftwareCost * (planDiscount / 100);
      const finalCost = totalSoftwareCost - planDiscountAmount;
      
      // Implementation
      const implCost = implementation !== 'none' ? implementations[implementation].price : 0;
      const implDiscountAmount = implCost * (implDiscount / 100);
      const finalImplCost = implCost - implDiscountAmount;
      
      // Calculate vs monthly savings
      let vsMonthly = 0;
      // Year 1
      vsMonthly += planPricing.monthly.year1 * users * 12;
      // Years 2+
      if (years > 1) {
        vsMonthly += planPricing.monthly.year2plus * users * 12 * (years - 1);
      }
      const monthlySavings = vsMonthly - totalSoftwareCost;
      
      return {
        termLabel: `${years} Year${years > 1 ? 's' : ''}`,
        softwareCost: finalCost,
        implementationCost: finalImplCost,
        years: years,
        planDiscountAmount,
        implDiscountAmount,
        monthlySavings,
        amortizedMonthly: (finalCost + finalImplCost) / (years * 12),
        isMonthly: false
      };
    }
  };

  const activeTerms = Object.entries(selectedTerms)
    .filter(([_, isSelected]) => isSelected)
    .map(([termKey, _]) => termKey);

  const formatCurrency = (amount) => {
    return new Intl.NumberFormat('en-US', {
      style: 'currency',
      currency: 'USD',
      minimumFractionDigits: 2
    }).format(amount);
  };

  return (
    <div className="min-h-screen" style={{
      background: 'linear-gradient(135deg, #0a0a0a 0%, #1a1a2e 50%, #16213e 100%)',
      fontFamily: "'Inter', 'Lato', -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif"
    }}>
      {/* Animated background grid */}
      <div className="fixed inset-0 opacity-10" style={{
        backgroundImage: 'linear-gradient(rgba(135, 90, 123, 0.2) 1px, transparent 1px), linear-gradient(90deg, rgba(135, 90, 123, 0.2) 1px, transparent 1px)',
        backgroundSize: '50px 50px'
      }}></div>

      <div className="relative max-w-7xl mx-auto p-6">
        {/* Header */}
        <div className="bg-gradient-to-br from-gray-900 to-gray-800 rounded-3xl shadow-2xl p-8 mb-8 border border-purple-500/20 relative overflow-hidden">
          {/* Animated glow effect */}
          <div className="absolute inset-0 bg-gradient-to-r from-purple-600/10 via-transparent to-cyan-600/10 animate-pulse"></div>
          
          <div className="relative z-10 flex items-center justify-between">
            <div className="flex items-center gap-5">
              <div className="relative">
                <div className="absolute inset-0 bg-purple-600 blur-xl opacity-50 animate-pulse"></div>
                <div className="relative flex items-center justify-center w-16 h-16 rounded-2xl shadow-2xl" style={{
                  background: 'linear-gradient(135deg, #875A7B 0%, #A24689 100%)'
                }}>
                  <Code className="w-8 h-8 text-white" />
                </div>
              </div>
              <div>
                <h1 className="text-5xl font-black mb-1 bg-gradient-to-r from-purple-400 via-pink-400 to-cyan-400 bg-clip-text text-transparent" style={{ 
                  fontFamily: "'Inter', sans-serif",
                  letterSpacing: '-0.02em'
                }}>
                  Odoo Quote Engine
                </h1>
                <p className="text-gray-400 text-lg flex items-center gap-2" style={{ fontFamily: "'Inter', sans-serif" }}>
                  <Zap className="w-4 h-4 text-cyan-400" />
                  Advanced pricing calculator for modern businesses
                </p>
              </div>
            </div>
            
            <div className="hidden lg:flex items-center gap-3">
              <div className="px-5 py-2 rounded-xl bg-purple-600/20 border border-purple-500/30">
                <span className="text-purple-300 text-sm font-bold">v2.0</span>
              </div>
            </div>
          </div>
        </div>

        {/* Main Grid Layout */}
        <div className="grid grid-cols-1 lg:grid-cols-3 gap-6 mb-8">
          {/* Configuration Panel */}
          <div className="lg:col-span-1">
            <div className="bg-gradient-to-br from-gray-900 to-gray-800 rounded-3xl shadow-2xl p-6 border border-purple-500/20 sticky top-6">
              <div className="flex items-center gap-3 mb-6 pb-4 border-b border-purple-500/20">
                <div className="w-10 h-10 rounded-xl flex items-center justify-center bg-gradient-to-br from-purple-600 to-pink-600">
                  <Cpu className="w-5 h-5 text-white" />
                </div>
                <h2 className="text-2xl font-bold text-white" style={{ fontFamily: "'Inter', sans-serif" }}>
                  Configuration
                </h2>
              </div>

              <div className="space-y-6">
                {/* Number of Users */}
                <div>
                  <label className="block text-sm font-bold text-gray-300 mb-3 flex items-center gap-2" style={{ fontFamily: "'Inter', sans-serif" }}>
                    <Users className="w-4 h-4" />
                    Number of Users
                  </label>
                  <input
                    type="number"
                    min="1"
                    value={users}
                    onChange={(e) => setUsers(Math.max(1, parseInt(e.target.value) || 1))}
                    className="w-full px-4 py-3 bg-gray-800/50 border-2 border-purple-500/30 rounded-xl text-white text-xl font-bold transition-all"
                    style={{
                      outline: 'none',
                      fontFamily: "'Inter', sans-serif"
                    }}
                    onFocus={(e) => e.target.style.borderColor = '#875A7B'}
                    onBlur={(e) => e.target.style.borderColor = 'rgba(139, 92, 246, 0.3)'}
                  />
                </div>

                {/* Plan Selection */}
                <div>
                  <label className="block text-sm font-bold text-gray-300 mb-3 flex items-center gap-2" style={{ fontFamily: "'Inter', sans-serif" }}>
                    <Package className="w-4 h-4" />
                    Plan Type
                  </label>
                  <div className="grid grid-cols-2 gap-3">
                    {['standard', 'custom'].map((planType) => (
                      <button
                        key={planType}
                        onClick={() => setPlan(planType)}
                        className="py-3 px-4 rounded-xl font-bold text-sm transition-all border-2"
                        style={{
                          background: plan === planType ? 'linear-gradient(135deg, #875A7B 0%, #A24689 100%)' : 'rgba(31, 41, 55, 0.5)',
                          color: plan === planType ? 'white' : '#9CA3AF',
                          borderColor: plan === planType ? '#875A7B' : 'rgba(139, 92, 246, 0.2)',
                          fontFamily: "'Inter', sans-serif"
                        }}
                      >
                        {planType.charAt(0).toUpperCase() + planType.slice(1)}
                      </button>
                    ))}
                  </div>
                </div>

                {/* Implementation Package */}
                <div>
                  <label className="block text-sm font-bold text-gray-300 mb-3 flex items-center gap-2" style={{ fontFamily: "'Inter', sans-serif" }}>
                    <Zap className="w-4 h-4" />
                    Implementation
                  </label>
                  <select
                    value={implementation}
                    onChange={(e) => setImplementation(e.target.value)}
                    className="w-full px-4 py-3 bg-gray-800/50 border-2 border-purple-500/30 rounded-xl text-white text-sm font-semibold appearance-none cursor-pointer transition-all"
                    style={{
                      outline: 'none',
                      fontFamily: "'Inter', sans-serif",
                      backgroundImage: `url("data:image/svg+xml,%3csvg xmlns='http://www.w3.org/2000/svg' fill='none' viewBox='0 0 20 20'%3e%3cpath stroke='%23875A7B' stroke-linecap='round' stroke-linejoin='round' stroke-width='2' d='M6 8l4 4 4-4'/%3e%3c/svg%3e")`,
                      backgroundPosition: 'right 0.75rem center',
                      backgroundRepeat: 'no-repeat',
                      backgroundSize: '1.25em 1.25em',
                      paddingRight: '2.5rem'
                    }}
                    onFocus={(e) => e.target.style.borderColor = '#875A7B'}
                    onBlur={(e) => e.target.style.borderColor = 'rgba(139, 92, 246, 0.3)'}
                  >
                    <option value="none">No Implementation</option>
                    <option value="express">Express - 4h - {formatCurrency(540)}</option>
                    <option value="starter">Starter - 25h - {formatCurrency(3600)}</option>
                    <option value="basic">Basic - 50h - {formatCurrency(7000)}</option>
                    <option value="standard">Standard - 100h - {formatCurrency(12500)}</option>
                    <option value="custom">Custom - 200h - {formatCurrency(25000)}</option>
                  </select>
                </div>
              </div>

              {/* Stats Display */}
              <div className="mt-6 pt-6 border-t border-purple-500/20">
                <div className="grid grid-cols-2 gap-4">
                  <div className="bg-purple-900/20 rounded-xl p-4 border border-purple-500/20">
                    <div className="text-xs text-purple-300 font-bold mb-1">Active Terms</div>
                    <div className="text-2xl font-black text-white">{activeTerms.length}</div>
                  </div>
                  <div className="bg-cyan-900/20 rounded-xl p-4 border border-cyan-500/20">
                    <div className="text-xs text-cyan-300 font-bold mb-1">Total Users</div>
                    <div className="text-2xl font-black text-white">{users}</div>
                  </div>
                </div>
              </div>
            </div>
          </div>

          {/* Term Selection Cards */}
          <div className="lg:col-span-2">
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-6">
              {[
                { key: 'monthly', label: 'Monthly', icon: '📅', color: 'from-blue-600 to-cyan-600' },
                { key: '1year', label: '1 Year', icon: '1️⃣', color: 'from-purple-600 to-pink-600' },
                { key: '2year', label: '2 Years', icon: '2️⃣', color: 'from-pink-600 to-rose-600' },
                { key: '3year', label: '3 Years', icon: '3️⃣', color: 'from-orange-600 to-red-600' },
                { key: '4year', label: '4 Years', icon: '4️⃣', color: 'from-green-600 to-emerald-600' },
                { key: '5year', label: '5 Years', icon: '5️⃣', color: 'from-teal-600 to-cyan-600' }
              ].map((term) => (
                <div
                  key={term.key}
                  className="bg-gradient-to-br from-gray-900 to-gray-800 rounded-2xl shadow-xl p-5 transition-all border-2"
                  style={{
                    borderColor: selectedTerms[term.key] ? '#875A7B' : 'rgba(139, 92, 246, 0.2)',
                    transform: selectedTerms[term.key] ? 'scale(1.02)' : 'scale(1)'
                  }}
                >
                  <div className="flex items-center justify-between mb-4">
                    <div className="flex items-center gap-3">
                      <div className={`w-10 h-10 rounded-xl flex items-center justify-center bg-gradient-to-br ${term.color} shadow-lg`}>
                        <span className="text-xl">{term.icon}</span>
                      </div>
                      <h3 className="text-lg font-bold text-white" style={{ fontFamily: "'Inter', sans-serif" }}>
                        {term.label}
                      </h3>
                    </div>
                    <button
                      onClick={() => setSelectedTerms({ ...selectedTerms, [term.key]: !selectedTerms[term.key] })}
                      className="w-12 h-12 rounded-xl flex items-center justify-center transition-all border-2 shadow-lg"
                      style={{
                        background: selectedTerms[term.key] ? `linear-gradient(135deg, #875A7B 0%, #A24689 100%)` : 'rgba(31, 41, 55, 0.5)',
                        borderColor: selectedTerms[term.key] ? '#875A7B' : 'rgba(139, 92, 246, 0.3)',
                        color: selectedTerms[term.key] ? 'white' : '#6B7280'
                      }}
                    >
                      {selectedTerms[term.key] ? '✓' : '+'}
                    </button>
                  </div>

                  <div className="space-y-3">
                    <div>
                      <label className="block text-xs font-bold text-gray-400 mb-2" style={{ fontFamily: "'Inter', sans-serif" }}>
                        Plan Discount (%)
                      </label>
                      <input
                        type="number"
                        min="0"
                        max="100"
                        step="0.1"
                        value={termDiscounts[term.key].plan}
                        onChange={(e) => setTermDiscounts({ 
                          ...termDiscounts, 
                          [term.key]: { ...termDiscounts[term.key], plan: Math.max(0, Math.min(100, parseFloat(e.target.value) || 0)) }
                        })}
                        className="w-full px-3 py-2 bg-gray-800/50 border border-purple-500/30 rounded-lg text-white font-semibold text-sm transition-all"
                        style={{
                          outline: 'none',
                          fontFamily: "'Inter', sans-serif"
                        }}
                        placeholder="0.0"
                        onFocus={(e) => e.target.style.borderColor = '#875A7B'}
                        onBlur={(e) => e.target.style.borderColor = 'rgba(139, 92, 246, 0.3)'}
                      />
                    </div>
                    <div>
                      <label className="block text-xs font-bold text-gray-400 mb-2" style={{ fontFamily: "'Inter', sans-serif" }}>
                        Implementation Discount (%)
                      </label>
                      <input
                        type="number"
                        min="0"
                        max="100"
                        step="0.1"
                        value={termDiscounts[term.key].impl}
                        onChange={(e) => setTermDiscounts({ 
                          ...termDiscounts, 
                          [term.key]: { ...termDiscounts[term.key], impl: Math.max(0, Math.min(100, parseFloat(e.target.value) || 0)) }
                        })}
                        className="w-full px-3 py-2 bg-gray-800/50 border border-purple-500/30 rounded-lg text-white font-semibold text-sm transition-all"
                        style={{
                          outline: 'none',
                          fontFamily: "'Inter', sans-serif"
                        }}
                        placeholder="0.0"
                        onFocus={(e) => e.target.style.borderColor = '#875A7B'}
                        onBlur={(e) => e.target.style.borderColor = 'rgba(139, 92, 246, 0.3)'}
                      />
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>
        </div>

        {/* Quote Comparison Table */}
        {activeTerms.length > 0 ? (
          <div className="bg-gradient-to-br from-gray-900 to-gray-800 rounded-3xl shadow-2xl p-8 border border-purple-500/20">
            <div className="flex items-center justify-between mb-6 pb-6 border-b border-purple-500/20">
              <div className="flex items-center gap-3">
                <div className="w-12 h-12 rounded-xl flex items-center justify-center bg-gradient-to-br from-purple-600 to-pink-600 shadow-lg">
                  <DollarSign className="w-6 h-6 text-white" />
                </div>
                <div>
                  <h2 className="text-3xl font-black text-white" style={{ fontFamily: "'Inter', sans-serif" }}>
                    Quote Comparison
                  </h2>
                  <p className="text-gray-400 text-sm">{users} users • {plan.charAt(0).toUpperCase() + plan.slice(1)} Plan</p>
                </div>
              </div>
            </div>

            <div className="overflow-x-auto">
              <table className="w-full">
                <thead>
                  <tr className="border-b-2 border-purple-500/30">
                    <th className="text-left py-4 px-6 font-black text-gray-300 text-sm uppercase tracking-wider" style={{ fontFamily: "'Inter', sans-serif" }}>
                      Item
                    </th>
                    {activeTerms.map((termKey) => {
                      const quote = calculateTermQuote(termKey, termDiscounts[termKey].plan, termDiscounts[termKey].impl);
                      return (
                        <th key={termKey} className="text-right py-4 px-6 font-black text-lg bg-gradient-to-r from-purple-400 to-pink-400 bg-clip-text text-transparent" style={{ fontFamily: "'Inter', sans-serif" }}>
                          {quote.termLabel}
                        </th>
                      );
                    })}
                  </tr>
                </thead>
                <tbody>
                  {/* Software Cost */}
                  <tr className="border-b border-purple-500/10 hover:bg-purple-900/10 transition-colors">
                    <td className="py-4 px-6 font-semibold text-gray-300" style={{ fontFamily: "'Inter', sans-serif" }}>
                      Software Cost
                      <div className="text-xs text-gray-500 font-normal">{plan.charAt(0).toUpperCase() + plan.slice(1)} Plan</div>
                    </td>
                    {activeTerms.map((termKey) => {
                      const quote = calculateTermQuote(termKey, termDiscounts[termKey].plan, termDiscounts[termKey].impl);
                      return (
                        <td key={termKey} className="text-right py-4 px-6 font-bold text-white text-lg" style={{ fontFamily: "'Inter', sans-serif" }}>
                          {formatCurrency(quote.softwareCost)}
                          {quote.isMonthly && <span className="text-sm text-gray-400">/mo</span>}
                        </td>
                      );
                    })}
                  </tr>

                  {/* Implementation */}
                  {implementation !== 'none' && (
                    <tr className="border-b border-purple-500/10 hover:bg-purple-900/10 transition-colors">
                      <td className="py-4 px-6 font-semibold text-gray-300" style={{ fontFamily: "'Inter', sans-serif" }}>
                        Implementation
                        <div className="text-xs text-gray-500 font-normal">
                          {implementation.charAt(0).toUpperCase() + implementation.slice(1)} - {implementations[implementation].hours}h
                        </div>
                      </td>
                      {activeTerms.map((termKey) => {
                        const quote = calculateTermQuote(termKey, termDiscounts[termKey].plan, termDiscounts[termKey].impl);
                        return (
                          <td key={termKey} className="text-right py-4 px-6 font-bold text-white text-lg" style={{ fontFamily: "'Inter', sans-serif" }}>
                            {formatCurrency(quote.implementationCost)}
                          </td>
                        );
                      })}
                    </tr>
                  )}

                  {/* Discount Row */}
                  {activeTerms.some(termKey => termDiscounts[termKey].plan > 0 || termDiscounts[termKey].impl > 0) && (
                    <tr className="border-b-2 border-green-500/30 hover:bg-green-900/10 transition-colors" style={{ backgroundColor: 'rgba(16, 185, 129, 0.05)' }}>
                      <td className="py-4 px-6 font-bold text-green-400" style={{ fontFamily: "'Inter', sans-serif" }}>
                        💰 Total Discount
                      </td>
                      {activeTerms.map((termKey) => {
                        const quote = calculateTermQuote(termKey, termDiscounts[termKey].plan, termDiscounts[termKey].impl);
                        const totalDiscount = quote.planDiscountAmount + quote.implDiscountAmount;
                        return (
                          <td key={termKey} className="text-right py-4 px-6 font-bold text-green-400 text-lg" style={{ fontFamily: "'Inter', sans-serif" }}>
                            {totalDiscount > 0 ? `-${formatCurrency(totalDiscount)}` : '-'}
                          </td>
                        );
                      })}
                    </tr>
                  )}

                  {/* Total */}
                  <tr className="hover:bg-purple-900/20 transition-colors" style={{ backgroundColor: 'rgba(139, 92, 246, 0.1)' }}>
                    <td className="py-5 px-6 font-black text-2xl bg-gradient-to-r from-purple-400 via-pink-400 to-cyan-400 bg-clip-text text-transparent" style={{ fontFamily: "'Inter', sans-serif" }}>
                      TOTAL
                    </td>
                    {activeTerms.map((termKey) => {
                      const quote = calculateTermQuote(termKey, termDiscounts[termKey].plan, termDiscounts[termKey].impl);
                      const total = quote.softwareCost + quote.implementationCost;
                      return (
                        <td key={termKey} className="text-right py-5 px-6 font-black text-2xl bg-gradient-to-r from-purple-400 to-pink-400 bg-clip-text text-transparent" style={{ fontFamily: "'Inter', sans-serif" }}>
                          {formatCurrency(total)}
                          {quote.isMonthly && <span className="text-sm text-gray-400">/mo</span>}
                        </td>
                      );
                    })}
                  </tr>

                  {/* Amortized Monthly */}
                  {activeTerms.some(termKey => termKey !== 'monthly') && (
                    <tr className="border-t-2 border-purple-500/20 hover:bg-gray-900/50 transition-colors">
                      <td className="py-4 px-6 font-semibold text-gray-400" style={{ fontFamily: "'Inter', sans-serif" }}>
                        Amortized Monthly
                        <div className="text-xs text-gray-600 font-normal">For budgeting</div>
                      </td>
                      {activeTerms.map((termKey) => {
                        const quote = calculateTermQuote(termKey, termDiscounts[termKey].plan, termDiscounts[termKey].impl);
                        if (quote.isMonthly) {
                          return <td key={termKey} className="text-right py-4 px-6 text-gray-600">-</td>;
                        }
                        return (
                          <td key={termKey} className="text-right py-4 px-6 font-semibold text-gray-300 text-lg" style={{ fontFamily: "'Inter', sans-serif" }}>
                            {formatCurrency(quote.amortizedMonthly)}/mo
                          </td>
                        );
                      })}
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
        ) : (
          <div className="bg-gradient-to-br from-gray-900 to-gray-800 border-2 border-yellow-500/30 rounded-3xl p-12 text-center">
            <div className="w-20 h-20 bg-yellow-900/30 rounded-full flex items-center justify-center mx-auto mb-4 border border-yellow-500/30">
              <Package className="w-10 h-10 text-yellow-400" />
            </div>
            <p className="text-yellow-300 text-xl font-bold mb-2" style={{ fontFamily: "'Inter', sans-serif" }}>
              No terms selected
            </p>
            <p className="text-gray-400">Select at least one contract term above to generate your quote</p>
          </div>
        )}
      </div>
    </div>
  );
};

export default OdooQuoteCalculator;
